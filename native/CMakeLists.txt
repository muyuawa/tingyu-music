cmake_minimum_required(VERSION 3.24)
project(MuyuAudio LANGUAGES C CXX)
include(FetchContent)
set(BUILD_CXXLIBS OFF CACHE BOOL "" FORCE)
set(BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(WITH_OGG OFF CACHE BOOL "" FORCE)
set(INSTALL_MANPAGES OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(FLAC_ARCHIVE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/flac-e94ff9f68b8e7dbd3e9f8b1ac18a8eca1914f181.tar.gz")
if(NOT EXISTS "${FLAC_ARCHIVE}")
  set(FLAC_ARCHIVE "https://github.com/xiph/flac/archive/e94ff9f68b8e7dbd3e9f8b1ac18a8eca1914f181.tar.gz")
endif()
FetchContent_Declare(flac URL "${FLAC_ARCHIVE}" URL_HASH SHA256=FF5FC898E8299558998F69A063805FC41CEB9291FE9CE872CC67F7046669D0B9 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(flac)
set(FFMPEG_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/vendor/ffmpeg-minimal")
if(NOT EXISTS "${FFMPEG_ROOT}/include/libavcodec/avcodec.h")
  message(FATAL_ERROR "Build the pinned LGPL FFmpeg source with scripts/build-ffmpeg.ps1; see vendor/ffmpeg-source.json")
endif()
add_library(MuyuAudio SHARED audio.cpp)
target_compile_features(MuyuAudio PRIVATE cxx_std_20)
target_compile_definitions(MuyuAudio PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN FLAC__NO_DLL _CRT_SECURE_NO_WARNINGS)
target_link_libraries(MuyuAudio PRIVATE FLAC ole32 uuid avrt propsys bcrypt)
if(MSVC)
  target_compile_options(MuyuAudio PRIVATE /W4 /utf-8 /EHsc)
endif()
target_include_directories(MuyuAudio PRIVATE "${FFMPEG_ROOT}/include")
target_link_libraries(MuyuAudio PRIVATE "${FFMPEG_ROOT}/bin/avformat.lib" "${FFMPEG_ROOT}/bin/avcodec.lib" "${FFMPEG_ROOT}/bin/avutil.lib" "${FFMPEG_ROOT}/bin/swresample.lib")
foreach(dll avformat-62 avcodec-62 avutil-60 swresample-6)
  add_custom_command(TARGET MuyuAudio POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${FFMPEG_ROOT}/bin/${dll}.dll" "$<TARGET_FILE_DIR:MuyuAudio>")
endforeach()
