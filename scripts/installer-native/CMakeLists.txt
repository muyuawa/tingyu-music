cmake_minimum_required(VERSION 3.24)
project(TingyuInstaller LANGUAGES CXX)
add_library(TingyuInstaller SHARED installer.cpp)
target_compile_features(TingyuInstaller PRIVATE cxx_std_17)
target_compile_definitions(TingyuInstaller PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(TingyuInstaller PRIVATE /utf-8 /W4 /EHsc)
target_link_libraries(TingyuInstaller PRIVATE msi ole32 shell32 uuid advapi32)
set_property(TARGET TingyuInstaller PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
