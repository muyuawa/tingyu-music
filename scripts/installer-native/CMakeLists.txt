cmake_minimum_required(VERSION 3.24)
project(TingyuInstaller LANGUAGES CXX)
add_library(TingyuInstaller SHARED installer.cpp)
add_executable(TingyuInstallerProbe installer.cpp)
target_compile_definitions(TingyuInstallerProbe PRIVATE INSTALLER_PROBE)
foreach(target TingyuInstaller TingyuInstallerProbe)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(${target} PRIVATE /utf-8 /W4 /EHsc)
  target_link_libraries(${target} PRIVATE msi ole32 shell32 uuid advapi32)
  set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endforeach()
